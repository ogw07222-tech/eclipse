cmake_minimum_required(VERSION 3.24)
project(EclipseEngine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(ECLIPSE_HEADLESS "Build with SDL dummy video for CI smoke tests" OFF)
option(ECLIPSE_TESTS "Build ECLIPSE's small dependency-free core tests" ON)

include(FetchContent)
# Keep SDL's window, renderer, events, keyboard and touch support. Enable other
# subsystems only when ECLIPSE actually gains a feature that consumes them.
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
foreach(feature AUDIO CAMERA DIALOG GPU HAPTIC JOYSTICK SENSOR VULKAN TEST_LIBRARY TESTS EXAMPLES INSTALL)
  set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
if(ECLIPSE_HEADLESS)
  set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
  set(SDL_X11 OFF CACHE BOOL "" FORCE)
  set(SDL_WAYLAND OFF CACHE BOOL "" FORCE)
endif()
FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.2.22 GIT_SHALLOW TRUE GIT_SUBMODULES "")
FetchContent_MakeAvailable(SDL3)
# The game ships PNG sheets only. stb is bundled with SDL_image, so no libpng,
# JPEG, AVIF, TIFF, WebP, SVG or font decoder dependencies are required.
foreach(format AVIF BMP GIF JPG JXL LBM PCX PNM QOI SVG TGA TIF WEBP XCF XPM XV)
  set(SDLIMAGE_${format} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDLIMAGE_PNG ON CACHE BOOL "" FORCE)
set(SDLIMAGE_PNG_SAVE OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
set(SDLIMAGE_BACKEND_WIC OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_BACKEND_IMAGEIO OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_TESTS OFF CACHE BOOL "" FORCE)
set(SDLIMAGE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3_image GIT_REPOSITORY https://github.com/libsdl-org/SDL_image.git GIT_TAG release-3.2.4 GIT_SHALLOW TRUE GIT_SUBMODULES "")
FetchContent_MakeAvailable(SDL3_image)
if(ANDROID)
  add_library(main SHARED src/main.cpp)
  set(TARGET_NAME main)
else()
  add_executable(eclipse_engine src/main.cpp)
  set(TARGET_NAME eclipse_engine)
endif()
target_link_libraries(${TARGET_NAME} PRIVATE SDL3::SDL3 SDL3_image::SDL3_image)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
  if(NOT APPLE)
    target_link_options(${TARGET_NAME} PRIVATE -Wl,--gc-sections)
  endif()
endif()
# Use the shared engine entry-point convention from SDL_main.h. Defining
# SDL_MAIN_USE_CALLBACKS=0 still enables callbacks, so do not define it at all.
if(NOT ANDROID)
  set(ECLIPSE_ASSET_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/Resources/Eclipse")
  if(NOT EXISTS "${ECLIPSE_ASSET_SOURCE}/knight_motion.png")
    message(FATAL_ERROR "ECLIPSE knight atlas is missing: ${ECLIPSE_ASSET_SOURCE}")
  endif()
  set(ECLIPSE_GAME_ASSETS ui/white_night_lance.png ui/white_night_motion.png title/eclipse.png title/cape.png ui/title_menu.png knight_motion.png knight_skill_motion.png mage_run.png mage_motion.png ui/attack_states.png ui/dash_states.png ui/swap_skill_states.png ui/joystick_arrows_new.png ui/settings_gear.png ui/settings_menu_panel.png ui/settings_window.png ui/enemy_bar_frame.png ui/execute_skull.png ui/knight_skill_stab.png ui/knight_skill_sweep.png ui/knight_skill_prep.png monsters/slime_green.png monsters/slime_red.png monsters/slime_purple.png)
  foreach(asset IN LISTS ECLIPSE_GAME_ASSETS)
    set_property(TARGET ${TARGET_NAME} APPEND PROPERTY LINK_DEPENDS "${ECLIPSE_ASSET_SOURCE}/${asset}")
    get_filename_component(directory "${asset}" DIRECTORY)
    add_custom_command(TARGET ${TARGET_NAME} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${TARGET_NAME}>/assets/${directory}"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ECLIPSE_ASSET_SOURCE}/${asset}" "$<TARGET_FILE_DIR:${TARGET_NAME}>/assets/${asset}")
  endforeach()
  # Optional forest PNG set; preserve folder layout and only copy files that exist.
  file(GLOB_RECURSE ECLIPSE_FOREST_PNGS RELATIVE "${ECLIPSE_ASSET_SOURCE}" "${ECLIPSE_ASSET_SOURCE}/forest/*.png" "${ECLIPSE_ASSET_SOURCE}/forest_v2/*.png" "${ECLIPSE_ASSET_SOURCE}/forest_v3/*.png" "${ECLIPSE_ASSET_SOURCE}/forest_entrance/*.png" "${ECLIPSE_ASSET_SOURCE}/demon_castle/**/*.png")
  foreach(asset IN LISTS ECLIPSE_FOREST_PNGS)
    get_filename_component(directory "${asset}" DIRECTORY)
    add_custom_command(TARGET ${TARGET_NAME} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${TARGET_NAME}>/assets/${directory}"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ECLIPSE_ASSET_SOURCE}/${asset}" "$<TARGET_FILE_DIR:${TARGET_NAME}>/assets/${asset}")
  endforeach()
  set_target_properties(${TARGET_NAME} PROPERTIES BUILD_RPATH "$ORIGIN")
  if(ECLIPSE_TESTS)
    enable_testing()
    add_executable(eclipse_core_tests tests/core_tests.cpp)
    add_test(NAME eclipse_core COMMAND eclipse_core_tests)
    add_test(NAME eclipse_png_render_smoke COMMAND eclipse_engine --smoke-test)
    set_tests_properties(eclipse_png_render_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
  endif()
endif()
